// ======================================
//  ADDON - PRETS BANCAIRES
//  Integration: propose un pret quand un joueur descend sous 0.
// ======================================
const LOAN={
  activeByPlayer:{},
  amount:400,
  interest:1.25,
  repayTurns:6,
};

function getLoanState(playerId){
  return LOAN.activeByPlayer[playerId]||null;
}

function openLoanPanel(player){
  const panel=document.getElementById('loan-panel');
  const sub=document.getElementById('loan-sub');
  if(!panel||!sub) return;
  sub.textContent=player.name+' peut emprunter '+LOAN.amount+'$ (a rembourser: '+Math.ceil(LOAN.amount*LOAN.interest)+'$ sur '+LOAN.repayTurns+' tours).';
  panel.dataset.playerId=player.id;
  panel.classList.add('open');
  panel.style.display='block';
}

function closeLoanPanel(){
  const panel=document.getElementById('loan-panel');
  if(!panel) return;
  panel.style.display='none';
  panel.classList.remove('open');
  panel.dataset.playerId='';
}

function acceptLoan(){
  const panel=document.getElementById('loan-panel');
  if(!panel) return;
  const id=panel.dataset.playerId;
  const p=G.players.find(x=>x.id===id);
  if(!p){ closeLoanPanel(); return; }
  const total=Math.ceil(LOAN.amount*LOAN.interest);
  LOAN.activeByPlayer[p.id]={remaining:total,turnsLeft:LOAN.repayTurns};
  p.money+=LOAN.amount;
  recordTxn(p,LOAN.amount,'Pret bancaire');
  toast('Pret accepte par '+p.name, 'i');
  closeLoanPanel();
  updateUI();
}

function denyLoan(){
  closeLoanPanel();
}

function tickLoans(){
  G.players.forEach(p=>{
    const st=getLoanState(p.id);
    if(!st) return;
    if(st.turnsLeft<=0){
      // Penalite si retard
      const fee=Math.ceil(st.remaining*.12);
      st.remaining+=fee;
      st.turnsLeft=2;
      toast('Retard de pret: +'+fee+'$ pour '+p.name, 'd');
    }
    const due=Math.ceil(st.remaining/Math.max(1,st.turnsLeft));
    const pay=Math.min(due,p.money);
    p.money-=pay;
    st.remaining-=pay;
    st.turnsLeft--;
    if(pay>0) recordTxn(p,-pay,'Remboursement pret');
    if(st.remaining<=0){
      delete LOAN.activeByPlayer[p.id];
      toast(p.name+' a fini de rembourser son pret', 'a');
    }
    checkBust(p);
  });
}

(function patchCheckBustLoan(){
  const orig=window.checkBust;
  if(typeof orig!=='function') return;
  window.checkBust=function(player){
    if(player.money<0 && !getLoanState(player.id)){
      openLoanPanel(player);
    }
    return orig(player);
  };
})();

(function patchDoNextTurnLoan(){
  const orig=window.doNextTurn;
  if(typeof orig!=='function') return;
  window.doNextTurn=function(){
    tickLoans();
    orig();
  };
})();

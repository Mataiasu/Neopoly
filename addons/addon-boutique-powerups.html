// ═══════════════════════════════════════
//  ADDON — BOUTIQUE DE POWER-UPS
//  Accessible via openShop() pendant le tour du joueur courant.
// ═══════════════════════════════════════
const SHOP_ITEMS = [
  {
    id:'dice_boost', icon:'⚡', name:'Booster de Dé', price:150,
    desc:'Relance les dés une fois ce tour et garde le meilleur.',
    fn(player) {
      SHOP_STATE.diceBoostActive = true;
      toast('⚡ Booster actif ! Ton prochain lancer sera optimisé.', 'a');
    }
  },
  {
    id:'anti_jail', icon:'🛡️', name:'Anti-Prison', price:200,
    desc:'Sortie de prison automatique au prochain emprisonnement.',
    fn(player) {
      player.antiJailItem = (player.antiJailItem || 0) + 1;
      toast('🛡️ Anti-prison activé pour '+player.name, 'a');
    }
  },
  {
    id:'ghost', icon:'👻', name:'Fantôme', price:250,
    desc:'Aucun loyer lors du prochain atterrissage.',
    fn(player) {
      player.ghostTurn = true;
      toast('👻 Mode Fantôme activé ! Aucun loyer au prochain atterrissage.', 'a');
    }
  },
  {
    id:'quick_build', icon:'🏗️', name:'Construction Rapide', price:300,
    desc:'Construis une maison gratuitement maintenant.',
    fn(player) {
      const myProps = G.board.filter(c => c.owner === player.id && c.t === 'property' && (c.houses || 0) < 5);
      if (!myProps.length) { toast('Aucune propriété constructible.', 'w'); return false; }
      const prop = myProps[0]; // Première propriété disponible
      prop.houses = (prop.houses || 0) + 1;
      toast('🏗️ Maison construite sur ' + prop.n + ' !', 'a');
      renderBoard();
    }
  },
  {
    id:'sabotage', icon:'💣', name:'Sabotage', price:350,
    desc:'Détruit une maison sur une propriété adverse.',
    fn(player) {
      const targets = G.board.filter(c => c.owner && c.owner !== player.id && c.houses > 0);
      if (!targets.length) { toast('Aucune propriété adverse avec maison.', 'w'); return false; }
      const opts = targets.map((c,i)=>`${i}: ${c.n} (${c.houses} 🏠)`).join('\n');
      const idx = parseInt(prompt('Quelle propriété saboter ?\n' + opts), 10);
      if (isNaN(idx) || !targets[idx]) { toast('Sabotage annulé.', 'i'); player.money += 350; recordTxn(player, 350, 'Remboursement sabotage'); return false; }
      targets[idx].houses--;
      log('💣 ' + player.name + ' sabote ' + targets[idx].n, 'd');
      toast('💣 Sabotage réussi sur ' + targets[idx].n + '!', 'd');
      renderBoard();
    }
  },
  {
    id:'tax_collect', icon:'💰', name:'Collecte d\'Impôts', price:100,
    desc:'Récupère 20$ sur chaque autre joueur actif.',
    fn(player) {
      const others = G.players.filter(p => !p.bust && p.id !== player.id);
      let total = 0;
      others.forEach(p => {
        const take = Math.min(20, p.money);
        p.money -= take;
        player.money += take;
        total += take;
        recordTxn(p, -take, 'Impôt de '+player.name);
        checkBust(p);
      });
      recordTxn(player, total, 'Collecte d\'impôts');
      toast('💰 ' + player.name + ' collecte ' + total + '$ d\'impôts!', 'a');
    }
  },
];

const SHOP_STATE = {
  diceBoostActive: false,
};

function openShop() {
  if (!G.started) { toast('Lance une partie d\'abord.', 'w'); return; }
  renderShopItems();
  document.getElementById('shop-modal-bg').classList.add('open');
}

function closeShop() {
  document.getElementById('shop-modal-bg').classList.remove('open');
}

function renderShopItems() {
  const player = G.players[G.cur];
  if (!player) return;
  document.getElementById('shop-balance').textContent = 'Budget : ' + player.money + '$';
  const list = document.getElementById('shop-items-list');
  list.innerHTML = SHOP_ITEMS.map(item => {
    const canAfford = player.money >= item.price;
    return `<div class="shop-item${canAfford ? '' : ' cant-afford'}">
      <span class="shop-item-icon">${item.icon}</span>
      <div class="shop-item-info">
        <div class="shop-item-name">${escapeHtml(item.name)}</div>
        <div class="shop-item-desc">${escapeHtml(item.desc)}</div>
      </div>
      <span class="shop-item-price">${item.price}$</span>
      <button class="btn btn-blue btn-xs shop-item-btn" ${canAfford ? '' : 'disabled'} onclick="buyShopItem('${item.id}')">Acheter</button>
    </div>`;
  }).join('');
}

function buyShopItem(itemId) {
  const item = SHOP_ITEMS.find(i => i.id === itemId);
  const player = G.players[G.cur];
  if (!item || !player) return;
  if (player.money < item.price) { toast('Pas assez d\'argent !', 'd'); return; }
  player.money -= item.price;
  recordTxn(player, -item.price, 'Shop: ' + item.name);
  const result = item.fn(player);
  if (result !== false) updateUI();
  renderShopItems(); // Refresh stock
}

// Patch calcRent : mode fantôme
(function patchCalcRentGhost() {
  const orig = window.calcRent;
  if (typeof orig !== 'function') return;
  window.calcRent = function(cell, player) {
    if (player.ghostTurn) {
      player.ghostTurn = false;
      toast('👻 Mode Fantôme ! Pas de loyer.', 'a');
      return 0;
    }
    return orig(cell, player);
  };
})();

// Patch doRoll : dice boost
(function patchDoRollShop() {
  const orig = window.doRoll;
  if (typeof orig !== 'function') return;
  window.doRoll = function() {
    if (!SHOP_STATE.diceBoostActive) { orig(); return; }
    // Simule deux lancers côté client et garde le meilleur total
    const r1a = Math.ceil(Math.random() * 6);
    const r1b = Math.ceil(Math.random() * 6);
    const r2a = Math.ceil(Math.random() * 6);
    const r2b = Math.ceil(Math.random() * 6);
    const sum1 = r1a + r1b;
    const sum2 = r2a + r2b;
    const best = sum1 >= sum2 ? [r1a, r1b] : [r2a, r2b];
    SHOP_STATE.diceBoostActive = false;
    // Override le résultat de dés dans G si supporté
    if (typeof G._forceRoll !== 'undefined') {
      G._forceRoll = best;
    }
    orig();
    toast('⚡ Booster : tu avais le choix entre ' + sum1 + ' et ' + sum2 + '!', 'a');
  };
})();

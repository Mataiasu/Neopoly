🤝 Addon — Alliances entre Joueurs

NOUVELLE MÉCANIQUE SOCIALE

Les joueurs peuvent former des alliances temporaires pour s'entraider. Une alliance offre des avantages mutuels, mais peut être trahie à tout moment.

🤝 Former une alliance : propose à un autre joueur. Il doit accepter.
💰 Loyer réduit de 50% : les alliés ne se paient que la moitié du loyer entre eux.
🗡️ Trahison : briser une alliance inflige une pénalité de 200$ au traître.
⏳ Durée : une alliance dure jusqu'à trahison ou fin de partie.
👑 Limite : un joueur ne peut avoir qu'une seule alliance active à la fois.

// ═══════════════════════════════════════
//  ADDON — ALLIANCES
//  Nécessite que G.players[i].alliance soit initialisé à null.
//  Intégration : appeler initAllianceSystem() après doStart(),
//  et renderAlliancePanel() dans updateUI().
// ═══════════════════════════════════════
const ALLIANCE_BREACH_PENALTY = 200;
const ALLIANCE_RENT_FACTOR    = 0.5; // alliés paient 50% du loyer

function getAllyOf(playerId) {
  const p = G.players.find(x => x.id === playerId);
  if (!p || !p.alliance) return null;
  return G.players.find(x => x.id === p.alliance) || null;
}

function proposeAlliance(fromId, toId) {
  const from = G.players.find(p => p.id === fromId);
  const to   = G.players.find(p => p.id === toId);
  if (!from || !to || from.bust || to.bust) return;
  if (from.alliance) { toast(from.name + ' a déjà une alliance.', 'w'); return; }
  if (to.alliance)   { toast(to.name + ' a déjà une alliance.',   'w'); return; }

  const accept = confirm(to.name + ', acceptes-tu une alliance avec ' + from.name + ' ?');
  if (!accept) { toast(to.name + ' refuse l\'alliance.', 'i'); return; }

  from.alliance = toId;
  to.alliance   = fromId;
  toast('🤝 Alliance formée : ' + from.name + ' & ' + to.name, 'a');
  log('🤝 Alliance : ' + from.name + ' ↔ ' + to.name, 'a');
  renderAlliancePanel();
  renderPlayers();
}

function breakAlliance(fromId) {
  const from = G.players.find(p => p.id === fromId);
  if (!from || !from.alliance) return;
  const to = G.players.find(p => p.id === from.alliance);

  from.money = Math.max(0, from.money - ALLIANCE_BREACH_PENALTY);
  recordTxn(from, -ALLIANCE_BREACH_PENALTY, 'Pénalité trahison');
  toast('🗡️ ' + from.name + ' trahit ' + (to?.name || '?') + ' (−' + ALLIANCE_BREACH_PENALTY + '$)', 'd');
  log('🗡️ Trahison de ' + from.name + ' — pénalité ' + ALLIANCE_BREACH_PENALTY + '$', 'd');

  if (to) to.alliance = null;
  from.alliance = null;
  checkBust(from);
  renderAlliancePanel();
  renderPlayers();
  updateUI();
}

function renderAlliancePanel() {
  const panel = document.getElementById('alliance-panel');
  const status = document.getElementById('alliance-status');
  const actions = document.getElementById('alliance-actions');
  if (!panel || !status || !actions) return;

  if (!G.started || G.players.length < 2) { panel.style.display = 'none'; return; }
  panel.style.display = '';

  const curP = G.players[G.cur];
  if (!curP) return;

  const ally = getAllyOf(curP.id);
  if (ally) {
    status.innerHTML = 'Alliance avec <b>' + escapeHtml(ally.emoji + ' ' + ally.name) + '</b> (loyers -50% mutuels)';
    actions.innerHTML =
      '<button class="btn btn-danger btn-sm" onclick="breakAlliance(\'' + curP.id + '\')">🗡️ Trahir (-' + ALLIANCE_BREACH_PENALTY + '$)</button>';
  } else {
    const targets = G.players.filter(p => !p.bust && p.id !== curP.id && !p.alliance);
    if (!targets.length) {
      status.textContent = 'Aucun joueur disponible pour une alliance.';
      actions.innerHTML = '';
      return;
    }
    status.textContent = 'Propose une alliance à :';
    actions.innerHTML = targets.map(p =>
      '<button class="btn btn-yellow btn-sm" onclick="proposeAlliance(\'' + curP.id + '\',\'' + p.id + '\')">' +
      escapeHtml(p.emoji + ' ' + p.name) + '</button>'
    ).join('');
  }

  // Badges sur les player-rows
  document.querySelectorAll('.player-row').forEach(row => {
    const pid = row.dataset.pid;
    if (!pid) return;
    const p = G.players.find(x => x.id === pid);
    row.classList.toggle('allied', !!(p && p.alliance));
    let badge = row.querySelector('.alliance-badge');
    if (p && p.alliance) {
      const allyP = getAllyOf(p.id);
      if (!badge) {
        badge = document.createElement('span');
        badge.className = 'alliance-badge';
        row.querySelector('.player-name')?.appendChild(badge);
      }
      if (badge) badge.textContent = '🤝 ' + (allyP?.name || '?');
    } else {
      badge?.remove();
    }
  });
}

// Patch calcRent : loyer réduit entre alliés
(function patchCalcRentAlliance() {
  const orig = window.calcRent;
  if (typeof orig !== 'function') return;
  window.calcRent = function(cell, player) {
    const base = orig(cell, player);
    if (!cell.owner || cell.owner === player.id) return base;
    const owner = G.players.find(p => p.id === cell.owner);
    if (!owner) return base;
    if (owner.alliance === player.id) {
      return Math.ceil(base * ALLIANCE_RENT_FACTOR);
    }
    return base;
  };
})();

// Patch doStart : initialise alliance à null pour chaque joueur
(function patchDoStartAlliance() {
  const orig = window.doStart;
  if (typeof orig !== 'function') return;
  window.doStart = function() {
    orig();
    G.players.forEach(p => { p.alliance = null; });
    renderAlliancePanel();
  };
})();

// Patch updateUI : rafraîchit le panel
(function patchUpdateUIAlliance() {
  const orig = window.updateUI;
  if (typeof orig !== 'function') return;
  window.updateUI = function() {
    orig();
    renderAlliancePanel();
  };
})();

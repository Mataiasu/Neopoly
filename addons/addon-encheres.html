NOUVELLE RÈGLE

// ═══════════════════════════════════════
//  ADDON — SYSTÈME D'ENCHÈRES
//  Déclenché après declineBuy() quand un joueur refuse d'acheter.
//  Intégration : dans declineBuy(), remplace closeBuyBar() par startAuction().
// ═══════════════════════════════════════
const AUCTION = {
  cell: null,
  bids: {},   // { playerId: amount }
  passed: new Set(),
  order: [],  // IDs des joueurs dans l'ordre d'enchère
  curIdx: 0,
};

function startAuction() {
  if (!G.pendingBuy) return;
  const cell = G.board[G.pendingBuy];
  if (!cell || cell.t !== 'property' || cell.owner != null) {
    closeBuyBar();
    return;
  }
  AUCTION.cell   = cell;
  AUCTION.bids   = {};
  AUCTION.passed = new Set();
  // Ordre : joueur actuel en premier, puis les autres
  AUCTION.order  = G.players.filter(p => !p.bust).map(p => p.id);
  const ci = AUCTION.order.indexOf(G.players[G.cur].id);
  if (ci > 0) AUCTION.order = [...AUCTION.order.slice(ci), ...AUCTION.order.slice(0, ci)];
  AUCTION.curIdx = 0;
  G.pendingBuy   = null;
  closeBuyBar();
  renderAuctionModal();
  document.getElementById('auction-modal-bg').classList.add('open');
  log('🔨 Enchères ouvertes pour ' + cell.n, 'w');
}

function renderAuctionModal() {
  const cell = AUCTION.cell;
  document.getElementById('auction-sub').textContent =
    cell.n + ' — Mise de départ : ' + cell.p + '$';

  const list = document.getElementById('auction-bids-list');
  const maxBid = Math.max(0, ...Object.values(AUCTION.bids));
  list.innerHTML = AUCTION.order.map(pid => {
    const p = G.players.find(x => x.id === pid);
    if (!p) return '';
    const bid = AUCTION.bids[pid];
    const hasPassed = AUCTION.passed.has(pid);
    const isWinner = bid === maxBid && bid > 0;
    const cur = AUCTION.order[AUCTION.curIdx];
    const isActive = cur === pid && !hasPassed;
    return `<div class="auction-bid-row${isWinner ? ' winner' : ''}">
      <span class="auction-bid-name">${p.emoji} ${escapeHtml(p.name)}${isActive ? ' ◀' : ''}</span>
      <span class="auction-bid-amount">${hasPassed ? '—' : (bid ? bid + '$' : '?')}</span>
    </div>`;
  }).join('');

  // Affiche le formulaire seulement pour le joueur actif (en local = tous voient)
  const confirmBtn = document.getElementById('auction-confirm-btn');
  const allDone = AUCTION.order.every(pid => AUCTION.bids[pid] != null || AUCTION.passed.has(pid));
  if (confirmBtn) confirmBtn.style.display = allDone ? '' : 'none';

  const inp = document.getElementById('auction-bid-input');
  if (inp) {
    inp.min = maxBid + 1;
    inp.placeholder = 'Mise min. ' + (maxBid + 1) + '$';
    inp.value = '';
  }
  const cur = G.players.find(x => x.id === AUCTION.order[AUCTION.curIdx]);
  document.getElementById('auction-title').textContent =
    '🔨 Enchères — ' + (cur ? cur.emoji + ' ' + cur.name + ', à toi !' : 'Terminé');
}

function submitAuctionBid() {
  const curId = AUCTION.order[AUCTION.curIdx];
  const p = G.players.find(x => x.id === curId);
  if (!p) return;
  const inp = document.getElementById('auction-bid-input');
  const amount = parseInt(inp.value, 10);
  const maxBid = Math.max(0, ...Object.values(AUCTION.bids));
  if (!amount || amount <= maxBid) {
    toast('Mise trop faible ! Minimum ' + (maxBid + 1) + '$', 'w');
    return;
  }
  if (amount > p.money) {
    toast(p.name + " n'a pas assez d'argent !", 'd');
    return;
  }
  AUCTION.bids[curId] = amount;
  advanceAuction();
}

function passAuctionBid() {
  const curId = AUCTION.order[AUCTION.curIdx];
  AUCTION.passed.add(curId);
  advanceAuction();
}

function advanceAuction() {
  // Trouver le prochain joueur non passé
  let next = (AUCTION.curIdx + 1) % AUCTION.order.length;
  let tries = 0;
  while (AUCTION.passed.has(AUCTION.order[next]) && tries < AUCTION.order.length) {
    next = (next + 1) % AUCTION.order.length;
    tries++;
  }
  const allDone = AUCTION.order.every(pid => AUCTION.bids[pid] != null || AUCTION.passed.has(pid));
  if (allDone || tries >= AUCTION.order.length) {
    AUCTION.curIdx = AUCTION.order.length; // EOF
  } else {
    AUCTION.curIdx = next;
  }
  renderAuctionModal();
}

function finalizeAuction() {
  const bids = AUCTION.bids;
  const maxBid = Math.max(0, ...Object.values(bids));
  if (maxBid <= 0) {
    toast("Aucune offre. La propriété reste à la banque.", 'i');
    closeAuctionModal();
    return;
  }
  const winnerId = Object.keys(bids).find(id => bids[id] === maxBid);
  const winner = G.players.find(p => p.id === winnerId);
  const cell = AUCTION.cell;
  winner.money -= maxBid;
  cell.owner = winner.id;
  recordTxn(winner, -maxBid, 'Enchère: ' + cell.n);
  toast('🏆 ' + winner.name + ' remporte ' + cell.n + ' pour ' + maxBid + '$!', 'a');
  log('🔨 ' + winner.name + ' achète ' + cell.n + ' aux enchères pour ' + maxBid + '$', 'a');
  renderBoard();
  updateUI();
  closeAuctionModal();
}

function closeAuctionModal() {
  document.getElementById('auction-modal-bg').classList.remove('open');
}

(function patchDeclineBuy() {
  const origDecline = window.declineBuy;
  if (typeof origDecline === 'function') {
    window.declineBuy = function() {
      if (G.pendingBuy != null) {
        const cell = G.board[G.pendingBuy];
        if (cell && cell.t === 'property' && cell.owner == null) {
          startAuction();
          return;
        }
      }
      origDecline();
    };
  }
})();

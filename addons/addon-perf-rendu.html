Addon - Optimisation Rendu Plateau

PATCH PERFORMANCE

Ce patch limite les re-rendus trop frequents via requestAnimationFrame et evite plusieurs mises a jour UI dans le meme tick. Le gameplay ne change pas.

Etape 1 : Ajoute ce bloc JS en fin de script principal de test1.html.
// ======================================
//  ADDON - OPTIMISATION RENDU
//  Integration: remplace les appels directs a renderBoard/updateUI
//  par scheduleRender()/scheduleUI() quand possible.
// ======================================
const PERF_RENDER={
  boardQueued:false,
  uiQueued:false,
  boardCount:0,
  uiCount:0,
};

function scheduleRender(){
  if(PERF_RENDER.boardQueued) return;
  PERF_RENDER.boardQueued=true;
  requestAnimationFrame(()=>{
    PERF_RENDER.boardQueued=false;
    PERF_RENDER.boardCount++;
    if(typeof window.renderBoard==='function') window.renderBoard();
  });
}

function scheduleUI(){
  if(PERF_RENDER.uiQueued) return;
  PERF_RENDER.uiQueued=true;
  requestAnimationFrame(()=>{
    PERF_RENDER.uiQueued=false;
    PERF_RENDER.uiCount++;
    if(typeof window.updateUI==='function') window.updateUI();
  });
}

(function patchBurstCalls(){
  // Intercepte les appels repetitifs provenant de loops rapides.
  const map=[
    ['renderBoard','scheduleRender'],
    ['updateUI','scheduleUI'],
  ];

  map.forEach(([name,scheduler])=>{
    const orig=window[name];
    if(typeof orig!=='function') return;
    window['__orig_'+name]=orig;
    window[name]=function(){
      // Si appel force: renderBoard(true) -> immediate
      if(arguments[0]===true) return orig.apply(this,arguments);
      return window[scheduler]();
    };
  });
})();

(function exposePerfDebug(){
  window.perfRenderStats=function(){
    return {
      boardFrames:PERF_RENDER.boardCount,
      uiFrames:PERF_RENDER.uiCount,
      boardQueued:PERF_RENDER.boardQueued,
      uiQueued:PERF_RENDER.uiQueued,
    };
  };
})();

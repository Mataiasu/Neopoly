⚡ Addon — Mode Blitz

NOUVEAU MODE DE JEU

Le Mode Blitz applique automatiquement un ensemble de règles accélérées dès le démarrage d'une partie.

🕐 Timer de 30s par tour — le joueur perd son tour si le chrono expire
💰 Argent de départ réduit à 1000$ — parties plus courtes et tendues
🏃 Max 30 tours — le joueur le plus riche gagne à la fin
⚡ Loyers × 1.5 — chaque loyer payé est majoré de 50%
🎲 Doubles = rejouer immédiatement (pas de limite de doubles avant prison)
🔔 Toast de compte à rebours visible toutes les 10s

// ═══════════════════════════════════════
//  ADDON — MODE BLITZ
//  Applique des règles speed-run au démarrage d'une partie.
//  Activation : appeler activateBlitzMode() avant doStart().
//  Ou ajouter un bouton "Mode Blitz" dans la modal de partie locale.
// ═══════════════════════════════════════
const BLITZ = {
  active: false,
  rentMultiplier: 1.5,
  startMoney: 1000,
  maxTurns: 30,
  timerSec: 30,
  unlimitedDoubles: true,
  _warnAt: [20, 10],
};

function activateBlitzMode() {
  BLITZ.active = true;
  document.body.classList.add('blitz-mode');
}

function deactivateBlitzMode() {
  BLITZ.active = false;
  document.body.classList.remove('blitz-mode');
}

// Patch appliqué au démarrage de la partie pour injecter les règles Blitz
(function patchDoStartBlitz() {
  const origDoStart = window.doStart;
  if (typeof origDoStart !== 'function') return;
  window.doStart = function() {
    if (BLITZ.active) {
      G.startingMoney   = BLITZ.startMoney;
      G.timerEnabled    = true;
      G.timerDuration   = BLITZ.timerSec;
      G.maxTurns        = BLITZ.maxTurns;
      G.autoEnd         = true;
      toast('⚡ Mode Blitz actif ! 30s/tour, 30 tours max.', 'w');
    }
    origDoStart();
  };
})();

// Patch du calcul de loyer pour appliquer le multiplicateur Blitz
(function patchRentCalcBlitz() {
  const origCalcRent = window.calcRent;
  if (typeof origCalcRent !== 'function') return;
  window.calcRent = function(cell, player) {
    const base = origCalcRent(cell, player);
    if (BLITZ.active) return Math.ceil(base * BLITZ.rentMultiplier);
    return base;
  };
})();

// Patch doubles : en Blitz, pas de limite de doubles avant emprisonnement
(function patchDoublesBlitz() {
  const origDoRoll = window.doRoll;
  if (typeof origDoRoll !== 'function') return;
  window.doRoll = function() {
    const prevDouble = G.doublesCount;
    origDoRoll();
    if (BLITZ.active && BLITZ.unlimitedDoubles) {
      // Reset le compteur de doubles pour éviter l'envoi en prison automatique
      if (G.doublesCount >= 3) G.doublesCount = 0;
    }
  };
})();

// Avertissement 10s restantes pendant le timer de tour
(function patchTimerWarningBlitz() {
  const origTickTimer = window.tickTimer;
  if (typeof origTickTimer !== 'function') return;
  window.tickTimer = function() {
    origTickTimer();
    if (!BLITZ.active) return;
    const rem = parseInt(document.getElementById('timer-num')?.textContent, 10) || 0;
    if (BLITZ._warnAt.includes(rem)) {
      if (!BLITZ['_warned_' + rem]) {
        BLITZ['_warned_' + rem] = true;
        toast('⏰ Blitz : ' + rem + 's restantes !', 'w');
      }
    } else {
      BLITZ._warnAt.forEach(t => { if (rem > t) delete BLITZ['_warned_' + t]; });
    }
  };
})();

// Bouton toggle facilement intégrable dans l'UI
function toggleBlitzMode() {
  BLITZ.active ? deactivateBlitzMode() : activateBlitzMode();
  toast(BLITZ.active ? '⚡ Mode Blitz activé' : 'Mode Blitz désactivé', BLITZ.active ? 'w' : 'i');
}

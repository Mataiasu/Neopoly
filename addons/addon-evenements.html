🌀 Addon — Événements Aléatoires

NOUVELLES CARTES / MÉCANIQUES

Chaque début de tour, il y a une chance (20% par défaut) qu'un événement global frappe tous les joueurs. Ces événements peuvent être positifs, négatifs ou stratégiques.

🌪️ Crise éco : tous les joueurs perdent 10% de leur argent
💹 Boom immobilier : tous les loyers ×2 pendant 3 tours
🔥 Incendie : un propriétaire aléatoire perd une maison
🎁 Jackpot fiscal : tous reçoivent 150$
🚧 Travaux : une case aléatoire est bloquée (pas de loyer) 2 tours
🔀 Redistribution : tous les joueurs échangent 100$ avec leur voisin gauche
💸 Taxe spéciale : chaque joueur paie 50$ par propriété possédée
🌟 Bonus départ : bonus départ doublé pendant 5 tours

// ═══════════════════════════════════════
//  ADDON — ÉVÉNEMENTS ALÉATOIRES
//  Intégration : appeler checkRandomEvent() au début de chaque tour,
//  dans doNextTurn() ou doRoll() selon la préférence.
//  Patch automatique (doNextTurn) ci-dessous.
// ═══════════════════════════════════════
const RANDOM_EVENTS = [
  {
    id:'crisis', icon:'🌪️', title:'Crise Économique',
    desc:'La bourse s\'effondre. Chacun perd 10% de sa fortune.',
    effect:'−10% pour tous les joueurs',
    fn(players){
      players.forEach(p=>{
        const loss=Math.ceil(p.money*.1);
        p.money-=loss;
        recordTxn(p,-loss,'Crise éco');
        checkBust(p);
      });
    }
  },
  {
    id:'boom', icon:'💹', title:'Boom Immobilier',
    desc:'Les loyers explosent ! ×2 pendant 3 tours.',
    effect:'+100% sur tous les loyers (3 tours)',
    fn(players){
      EVENTS_STATE.rentMultiplierTurns=3;
      EVENTS_STATE.rentMultiplier=2;
    }
  },
  {
    id:'fire', icon:'🔥', title:'Incendie !',
    desc:'Un immeuble prend feu. Un propriétaire perd une maison.',
    effect:'Destruction d\'une maison aléatoire',
    fn(players){
      const withHouse=[];
      G.board.forEach(c=>{if(c.houses>0)withHouse.push(c);});
      if(!withHouse.length)return;
      const cell=withHouse[Math.floor(Math.random()*withHouse.length)];
      cell.houses--;
      log('🔥 Incendie ! '+cell.n+' perd une maison.','d');
    }
  },
  {
    id:'jackpot', icon:'🎁', title:'Jackpot Fiscal',
    desc:'L\'État redistribue de l\'argent. Chacun reçoit 150$.',
    effect:'+150$ pour tous les joueurs',
    fn(players){
      players.forEach(p=>{p.money+=150;recordTxn(p,150,'Jackpot fiscal');});
    }
  },
  {
    id:'works', icon:'🚧', title:'Travaux',
    desc:'Des travaux bloquent une rue. Pas de loyer dessus pendant 2 tours.',
    effect:'Case aléatoire sans loyer (2 tours)',
    fn(players){
      const props=G.board.filter(c=>c.t==='property'&&c.owner!=null);
      if(!props.length)return;
      const cell=props[Math.floor(Math.random()*props.length)];
      EVENTS_STATE.blockedCells[cell.i]=(EVENTS_STATE.blockedCells[cell.i]||0)+2;
      log('🚧 Travaux sur '+cell.n+' — loyer suspendu 2 tours.','w');
    }
  },
  {
    id:'redistrib', icon:'🔀', title:'Redistribution',
    desc:'La conjoncture mélange les fortunes. Chacun donne 100$ à son voisin.',
    effect:'Rotation de 100$ entre joueurs',
    fn(players){
      const alive=players.filter(p=>!p.bust);
      const saved=alive.map(p=>Math.min(100,p.money));
      alive.forEach((p,i)=>{
        p.money-=saved[i];
        const next=alive[(i+1)%alive.length];
        next.money+=saved[i];
        recordTxn(p,-saved[i],'Redistribution');
        recordTxn(next,saved[i],'Redistribution reçue');
      });
    }
  },
  {
    id:'special_tax', icon:'💸', title:'Taxe Spéciale',
    desc:'Le gouvernement impose chaque propriété. 50$ par bien possédé.',
    effect:'−50$ × nb propriétés par joueur',
    fn(players){
      players.forEach(p=>{
        const count=G.board.filter(c=>c.owner===p.id).length;
        if(!count)return;
        const tax=count*50;
        p.money-=tax;
        recordTxn(p,-tax,'Taxe spéciale');
        checkBust(p);
      });
    }
  },
  {
    id:'double_go', icon:'🌟', title:'Bonus Départ Doublé',
    desc:'Les caisses débordent ! Le bonus départ est doublé pendant 5 tours.',
    effect:'Bonus Départ ×2 (5 tours)',
    fn(players){
      EVENTS_STATE.goMultiplierTurns=5;
      EVENTS_STATE.goMultiplier=2;
    }
  },
];

const EVENTS_STATE={
  chance:0.20,          // probabilité d'événement par tour
  rentMultiplier:1,
  rentMultiplierTurns:0,
  goMultiplier:1,
  goMultiplierTurns:0,
  blockedCells:{},      // {cellIndex: turnsLeft}
};

function checkRandomEvent(){
  if(!G.started)return;
  if(Math.random()>EVENTS_STATE.chance)return;
  const ev=RANDOM_EVENTS[Math.floor(Math.random()*RANDOM_EVENTS.length)];
  const alive=G.players.filter(p=>!p.bust);
  ev.fn(alive);
  showEventPopup(ev);
  updateUI();
}

function showEventPopup(ev){
  document.getElementById('event-popup-icon').textContent=ev.icon;
  document.getElementById('event-popup-title').textContent=ev.title;
  document.getElementById('event-popup-desc').textContent=ev.desc;
  document.getElementById('event-popup-effect').textContent=ev.effect;
  document.getElementById('event-popup').classList.add('open');
}

function closeEventPopup(){
  document.getElementById('event-popup').classList.remove('open');
}

// Décrémenter les effets temporaires chaque tour
function tickEventStates(){
  if(EVENTS_STATE.rentMultiplierTurns>0){
    EVENTS_STATE.rentMultiplierTurns--;
    if(EVENTS_STATE.rentMultiplierTurns<=0)EVENTS_STATE.rentMultiplier=1;
  }
  if(EVENTS_STATE.goMultiplierTurns>0){
    EVENTS_STATE.goMultiplierTurns--;
    if(EVENTS_STATE.goMultiplierTurns<=0)EVENTS_STATE.goMultiplier=1;
  }
  Object.keys(EVENTS_STATE.blockedCells).forEach(ci=>{
    EVENTS_STATE.blockedCells[ci]--;
    if(EVENTS_STATE.blockedCells[ci]<=0)delete EVENTS_STATE.blockedCells[ci];
  });
}

// Patch doNextTurn : déclenche l'événement en début de nouveau tour
(function patchDoNextTurnEvents(){
  const orig=window.doNextTurn;
  if(typeof orig!=='function')return;
  window.doNextTurn=function(){
    tickEventStates();
    checkRandomEvent();
    orig();
  };
})();

// Patch calcRent : applique le multiplicateur d'événement et les blocages
(function patchCalcRentEvents(){
  const orig=window.calcRent;
  if(typeof orig!=='function')return;
  window.calcRent=function(cell,player){
    if(EVENTS_STATE.blockedCells[cell.i]>0)return 0; // case bloquée par travaux
    const base=orig(cell,player);
    return Math.ceil(base*EVENTS_STATE.rentMultiplier);
  };
})();

// Patch collectPassGo : applique le bonus départ doublé
(function patchPassGoEvents(){
  const orig=window.collectPassGo;
  if(typeof orig!=='function')return;
  window.collectPassGo=function(player){
    if(EVENTS_STATE.goMultiplierTurns>0&&EVENTS_STATE.goMultiplier>1){
      player.money+=G.passGoBonus*(EVENTS_STATE.goMultiplier-1);
      recordTxn(player,G.passGoBonus*(EVENTS_STATE.goMultiplier-1),'Bonus Départ ×'+EVENTS_STATE.goMultiplier);
    }
    orig(player);
  };
})();

Addon - Saisons Dynamiques

NOUVELLE MECANIQUE

Le jeu change de saison tous les 8 tours. Chaque saison modifie les regles economiques: loyers, bonus depart et cout des maisons.

Printemps: bonus depart +25%
Ete: loyers +20%
Automne: maisons -20%
Hiver: loyers -15% et impots +15%

// ======================================
//  ADDON - SAISONS DYNAMIQUES
//  Integration: change les multiplicateurs selon le tour.
// ======================================
const SEASONS=[
  {id:'spring',label:'Printemps',rent:1,go:1.25,house:1,tax:1,chip:'spring'},
  {id:'summer',label:'Ete',rent:1.2,go:1,house:1,tax:1,chip:'summer'},
  {id:'autumn',label:'Automne',rent:1,go:1,house:.8,tax:1,chip:'autumn'},
  {id:'winter',label:'Hiver',rent:.85,go:1,house:1,tax:1.15,chip:'winter'},
];

const SEASON_STATE={every:8,index:0};

function getSeason(){ return SEASONS[SEASON_STATE.index]; }

function renderSeasonChip(){
  const el=document.getElementById('season-chip');
  if(!el) return;
  const s=getSeason();
  el.textContent='Saison: '+s.label;
  el.classList.remove('summer','autumn','winter');
  if(s.chip!=='spring') el.classList.add(s.chip);
}

function updateSeasonByTurn(){
  const turn=Math.max(1, G.turn || 1);
  const idx=Math.floor((turn-1)/SEASON_STATE.every)%SEASONS.length;
  if(idx!==SEASON_STATE.index){
    SEASON_STATE.index=idx;
    const s=getSeason();
    toast('Saison active: '+s.label, 'i');
    log('Meteo economique -> '+s.label, 'i');
  }
  renderSeasonChip();
}

(function patchDoNextTurnSeason(){
  const orig=window.doNextTurn;
  if(typeof orig!=='function') return;
  window.doNextTurn=function(){
    orig();
    updateSeasonByTurn();
  };
})();

(function patchCalcRentSeason(){
  const orig=window.calcRent;
  if(typeof orig!=='function') return;
  window.calcRent=function(cell,player){
    const base=orig(cell,player);
    return Math.ceil(base*getSeason().rent);
  };
})();

(function patchCollectPassGoSeason(){
  const orig=window.collectPassGo;
  if(typeof orig!=='function') return;
  window.collectPassGo=function(player){
    const before=player.money;
    orig(player);
    const bonus=player.money-before;
    const extra=Math.round(bonus*(getSeason().go-1));
    if(extra>0){
      player.money+=extra;
      recordTxn(player,extra,'Bonus saison '+getSeason().label);
    }
  };
})();

(function patchGetHouseCostSeason(){
  const orig=window.getHouseCost;
  if(typeof orig!=='function') return;
  window.getHouseCost=function(cell){
    return Math.ceil(orig(cell)*getSeason().house);
  };
})();
